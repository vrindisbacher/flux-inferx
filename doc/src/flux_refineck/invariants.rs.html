23pub fn check_invariants<'genv, 'tcx>(
24    genv: GlobalEnv<'genv, 'tcx>,
25    cache: &mut FixQueryCache,
26    def_id: MaybeExternId,
27    invariants: &[fhir::Expr],
28    adt_def: &rty::AdtDef,
29    def_id_to_cstr: &mut FxIndexMap<DefId, Task<FixpointTypes>>,
30    def_id_to_fixpoint_ctx: &mut FxIndexMap<DefId, FixpointCtxt<'genv, 'tcx, Tag>>,
31) -> Result<(), ErrorGuaranteed> {
32    // FIXME(nilehmann) maybe we should record whether the invariants were generated with overflow
33    // checking enabled and only assume them in code that also overflow checking enabled.
34    // Although, enable overflow checking locally is unsound in general.
35    //
36    // The good way would be to make overflow checking a property of a type that can be turned on
37    // and off locally. Then we consider an overflow-checked `T` distinct from a non-checked one and
38    // error/warn in case of a mismatch: overflow-checked types can flow to non-checked code but not
39    // the other way around.
40    let opts = genv.infer_opts(def_id.local_id());
41    adt_def
42        .invariants()
43        .iter_identity()
44        .enumerate()
45        .try_for_each_exhaust(|(idx, invariant)| {
46            let span = invariants[idx].span;
47            check_invariant(
48                genv,
49                cache,
50                def_id,
51                adt_def,
52                span,
53                invariant,
54                opts,
55                def_id_to_cstr,
56                def_id_to_fixpoint_ctx,
57            )
58        })
59}

61fn check_invariant<'genv, 'tcx>(
62    genv: GlobalEnv<'genv, 'tcx>,
63    cache: &mut FixQueryCache,
64    def_id: MaybeExternId,
65    adt_def: &rty::AdtDef,
66    span: Span,
67    invariant: &rty::Invariant,
68    opts: InferOpts,
69    def_id_to_cstr: &mut FxIndexMap<DefId, Task<FixpointTypes>>,
70    def_id_to_fixpoint_ctx: &mut FxIndexMap<DefId, FixpointCtxt<'genv, 'tcx, Tag>>,
71) -> Result<(), ErrorGuaranteed> {
72    let resolved_id = def_id.resolved_id();
73
74    let region_infercx = genv
75        .tcx()
76        .infer_ctxt()
77        .with_next_trait_solver(true)
78        .build(TypingMode::non_body_analysis());
79
80    let mut infcx_root = try_query(|| {
81        genv.infcx_root(&region_infercx, opts)
82            .identity_for_item(resolved_id)?
83            .build()
84    })
85    .emit(&genv)?;
86
87    for variant_idx in adt_def.variants().indices() {
88        let mut rcx = infcx_root.infcx(resolved_id, &region_infercx);
89
90        let variant_sig = genv
91            .variant_sig(adt_def.did(), variant_idx)
92            .emit(&genv)?
93            .expect("cannot check opaque structs")
94            .instantiate_identity()
95            .replace_bound_refts_with(|sort, _, kind| {
96                rty::Expr::fvar(rcx.define_bound_reft_var(sort, kind))
97            });
98
99        for ty in variant_sig.fields() {
100            let ty = rcx.unpack(ty);
101            rcx.assume_invariants(&ty);
102        }
103        let pred = invariant.apply(&variant_sig.idx);
104        rcx.check_pred(&pred, Tag::new(ConstrReason::Other, DUMMY_SP));
105    }
106    let answer = infcx_root
107        .execute_fixpoint_query(
108            cache,
109            def_id,
110            FixpointQueryKind::Invariant,
111            def_id_to_cstr,
112            def_id_to_fixpoint_ctx,
113        )
114        .emit(&genv)?;
115
116    if answer.errors.is_empty() {
117        Ok(())
118    } else {
119        Err(genv.sess().emit_err(errors::Invalid { span }))
120    }
121}
